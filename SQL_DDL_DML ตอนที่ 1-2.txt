AS

    p.ProductID,
    p.ProductName,
    c.CategoryName,
    s.CompanyName AS SupplierCompanyName,
    p.UnitPrice

ระดับ 3 · ท้าทาย

ข้อ 1.13 สร้างตาราง Training.PurchaseLine ตามข้อกำหนด
------------------------------------------------------------
CREATE TABLE Training.PurchaseLine
(
    PurchaseLineID int IDENTITY(1,1) NOT NULL,
    SupplierID     int NOT NULL,
    ProductID      int NOT NULL,
    Quantity       smallint NOT NULL,
    UnitCost       money NOT NULL,
    Discount       real NOT NULL
        CONSTRAINT DF_PurchaseLine_Discount DEFAULT (0),

    LineTotal AS
    (
        CONVERT(money, Quantity * UnitCost * (1.0 - Discount))
    ) PERSISTED,

    CONSTRAINT PK_PurchaseLine_PurchaseLineID
        PRIMARY KEY (PurchaseLineID),

    CONSTRAINT FK_PurchaseLine_SupplierID
        FOREIGN KEY (SupplierID)
        REFERENCES Training.LocalSupplier (SupplierID)
        ON DELETE CASCADE,

    CONSTRAINT FK_PurchaseLine_ProductID
        FOREIGN KEY (ProductID)
        REFERENCES dbo.Products (ProductID),

    CONSTRAINT CK_PurchaseLine_Quantity
        CHECK (Quantity > 0),

    CONSTRAINT CK_PurchaseLine_UnitCost
        CHECK (UnitCost >= 0),

    CONSTRAINT CK_PurchaseLine_Discount
        CHECK (Discount >= 0 AND Discount <= 1),

    CONSTRAINT UQ_PurchaseLine_SupplierID_ProductID
        UNIQUE (SupplierID, ProductID)
);
GO

คำอธิบาย:
- IDENTITY ทำให้ PurchaseLineID เพิ่มค่าอัตโนมัติ
- ON DELETE CASCADE ที่ SupplierID ทำให้เมื่อลบผู้ขาย รายการซื้อของผู้ขายนั้นถูกลบตาม
- Foreign key ของ ProductID ไม่กำหนด CASCADE จึงใช้พฤติกรรม NO ACTION ตามค่าเริ่มต้น ป้องกันการลบสินค้าที่ยังถูกอ้างอิง
- LineTotal เป็น computed column แบบ PERSISTED จึงเก็บค่าที่คำนวณไว้จริง
- UNIQUE (SupplierID, ProductID) ป้องกันสินค้าซ้ำภายในผู้ขายรายเดียวกัน

ข้อ 1.15 สร้างมุมมองสรุปยอดซื้อและจำนวนรายการ พร้อม Clustered Index
------------------------------------------------------------
-- ตั้งค่า SET options ที่จำเป็นสำหรับ Indexed View
SET ANSI_NULLS ON;
SET QUOTED_IDENTIFIER ON;
SET ANSI_PADDING ON;
SET ANSI_WARNINGS ON;
SET CONCAT_NULL_YIELDS_NULL ON;
SET ARITHABORT ON;
SET NUMERIC_ROUNDABORT OFF;
GO

CREATE VIEW Training.vw_SupplierSpend
WITH SCHEMABINDING
AS
SELECT
    pl.SupplierID,
    COUNT_BIG(*) AS PurchaseLineCount,
    SUM(pl.LineTotal) AS TotalSpend
FROM Training.PurchaseLine AS pl
GROUP BY pl.SupplierID;
GO

CREATE UNIQUE CLUSTERED INDEX IX_vw_SupplierSpend
ON Training.vw_SupplierSpend (SupplierID);
GO

ข้อ 2.15 ใช้ MERGE ปรับ SupplierProduct ของ SupplierID = 1 ให้ตรงกับสินค้าที่ยังไม่เลิกจำหน่าย
------------------------------------------------------------
MERGE Training.SupplierProduct AS target
USING
(
    SELECT
        ProductID,
        UnitPrice
    FROM dbo.Products
    WHERE Discontinued = 0
) AS source
ON target.SupplierID = 1
AND target.ProductID = source.ProductID

WHEN MATCHED THEN
    UPDATE SET
        target.UnitCost = source.UnitPrice

WHEN NOT MATCHED BY TARGET THEN
    INSERT (SupplierID, ProductID, UnitCost)
    VALUES (1, source.ProductID, source.UnitPrice)

WHEN NOT MATCHED BY SOURCE
     AND target.SupplierID = 1 THEN
    DELETE
;
GO
