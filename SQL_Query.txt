ข้อ 18 นับจำนวนนักศึกษาในแต่ละสาขา เรียงจากมากไปน้อย

ข้อ 19 หาจำนวนตอนเรียนและจำนวนที่นั่งรวมในแต่ละภาคการศึกษา

ข้อ 20 นับจำนวนรายการลงทะเบียนแยกตามเกรด ไม่นับรายการที่ยังไม่มีเกรด เรียงตามเกรด

ข้อ 26 แสดงตอนเรียนภาค 2568/1 พร้อมชื่อวิชา ชื่ออาจารย์ และอาคาร

ข้อ 27 แสดงรายวิชาที่นักศึกษาชื่อ กิตติ ลงทะเบียน พร้อมภาคการศึกษาและเกรด

ข้อ 28 แสดงนักศึกษาทุกคนพร้อมจำนวนตอนเรียนที่ลงทะเบียน รวมคนที่ไม่ได้ลงทะเบียนด้วย

ข้อ 31 นับจำนวนนักศึกษาที่ลงทะเบียนในแต่ละตอนเรียน พร้อมชื่อวิชา เรียงตาม section_id

ข้อ 32 หาจำนวนรายวิชาและหน่วยกิตรวมที่แต่ละสาขาเปิดสอน แสดงชื่อสาขา รวมสาขาที่ไม่มีวิชาด้วย

ข้อ 39 แสดงนักศึกษาที่เคยได้เกรด F อย่างน้อย 1 วิชา โดยใช้ EXISTS

ข้อ 40 แสดงนักศึกษาที่อายุน้อยที่สุดในแต่ละสาขา โดยใช้ Correlated subquery
SELECT d.dept_name,
       s.first_name,
       s.last_name,
       s.birth_date
FROM STUDENT AS s
INNER JOIN DEPARTMENT AS d
    ON s.dept_id = d.dept_id
WHERE s.birth_date = (
    SELECT MAX(s2.birth_date)
    FROM STUDENT AS s2
    WHERE s2.dept_id = s.dept_id
);


ข้อ 41 แสดงตอนเรียนที่มีนักศึกษาลงทะเบียนมากที่สุด (กรณีเสมอแสดงทุกตอน)
WITH section_counts AS (
    SELECT sec.section_id,
           COUNT(e.student_id) AS student_count
    FROM SECTION AS sec
    LEFT JOIN ENROLLMENT AS e
        ON sec.section_id = e.section_id
    GROUP BY sec.section_id
)
SELECT section_id, student_count
FROM section_counts
WHERE student_count = (
    SELECT MAX(student_count)
    FROM section_counts
);


ข้อ 42 แปลงเกรดเป็นแต้มด้วย CASE เฉพาะรายการที่มีเกรดของ section 4
SELECT student_id,
       section_id,
       grade,
       CASE grade
           WHEN 'A'  THEN 4
           WHEN 'B+' THEN 3.5
           WHEN 'B'  THEN 3
           WHEN 'C+' THEN 2.5
           WHEN 'C'  THEN 2
           WHEN 'D+' THEN 1.5
           WHEN 'D'  THEN 1
           WHEN 'F'  THEN 0
       END AS grade_point
FROM ENROLLMENT
WHERE section_id = 4
  AND grade IS NOT NULL;


ข้อ 43 จัดกลุ่มห้องเรียนตามขนาด แล้วนับจำนวนห้องในแต่ละกลุ่ม
SELECT
    CASE
        WHEN capacity < 35 THEN 'เล็ก'
        WHEN capacity BETWEEN 35 AND 60 THEN 'กลาง'
        ELSE 'ใหญ่'
    END AS room_size,
    COUNT(*) AS room_count
FROM ROOM
GROUP BY
    CASE
        WHEN capacity < 35 THEN 'เล็ก'
        WHEN capacity BETWEEN 35 AND 60 THEN 'กลาง'
        ELSE 'ใหญ่'
    END;


ข้อ 44 ใช้ CTE คำนวณ GPA ถ่วงด้วยหน่วยกิตของนักศึกษาแต่ละคน เฉพาะวิชาที่มีเกรด ปัด 2 ตำแหน่ง
WITH grade_points AS (
    SELECT e.student_id,
           c.credits,
           CASE e.grade
               WHEN 'A'  THEN 4.0
               WHEN 'B+' THEN 3.5
               WHEN 'B'  THEN 3.0
               WHEN 'C+' THEN 2.5
               WHEN 'C'  THEN 2.0
               WHEN 'D+' THEN 1.5
               WHEN 'D'  THEN 1.0
               WHEN 'F'  THEN 0.0
           END AS grade_point
    FROM ENROLLMENT AS e
    INNER JOIN SECTION AS sec
        ON e.section_id = sec.section_id
    INNER JOIN COURSE AS c
        ON sec.course_id = c.course_id
    WHERE e.grade IS NOT NULL
)
SELECT student_id,
       ROUND(SUM(grade_point * credits) / SUM(credits), 2) AS GPA
FROM grade_points
GROUP BY student_id
ORDER BY GPA DESC;


ข้อ 45 ใช้ CTE หาสาขาที่มีจำนวนนักศึกษามากกว่าค่าเฉลี่ยจำนวนนักศึกษาต่อสาขา
WITH department_counts AS (
    SELECT d.dept_id,
           d.dept_name,
           COUNT(s.student_id) AS student_count
    FROM DEPARTMENT AS d
    LEFT JOIN STUDENT AS s
        ON d.dept_id = s.dept_id
    GROUP BY d.dept_id, d.dept_name
)
SELECT dept_id, dept_name, student_count
FROM department_counts
WHERE student_count > (
    SELECT AVG(student_count)
    FROM department_counts
);


ข้อ 46 ใช้ RANK() จัดอันดับห้องเรียนตามความจุจากมากไปน้อย ภายในแต่ละอาคาร
SELECT room_id,
       building,
       capacity,
       RANK() OVER (
           PARTITION BY building
           ORDER BY capacity DESC
       ) AS room_rank

ข้อ 47 แสดงรายการลงทะเบียนของ section 2 และ 4 พร้อมจำนวนนักศึกษาทั้งหมดในตอนนั้น โดยใช้ COUNT OVER
SELECT student_id,
       section_id,
       enroll_date,
       grade,
       COUNT(*) OVER (
           PARTITION BY section_id
       ) AS section_student_count
FROM ENROLLMENT
WHERE section_id IN (2, 4);


ข้อ 48 ใช้ ROW_NUMBER() หารายการลงทะเบียนครั้งแรกของนักศึกษาแต่ละคน ตาม enroll_date
WITH ranked_enrollments AS (
    SELECT student_id,
           section_id,
           enroll_date,
           grade,
           ROW_NUMBER() OVER (
               PARTITION BY student_id
               ORDER BY enroll_date ASC, section_id ASC
           ) AS row_num
    FROM ENROLLMENT
)
SELECT student_id, section_id, enroll_date, grade
FROM ranked_enrollments
WHERE row_num = 1;

ข้อ 50 สรุปผลรายวิชาที่เปิดในภาค 2568/1
แสดงชื่อวิชา จำนวนผู้ลงทะเบียน จำนวนที่ได้ A อัตราการสอบผ่าน (%) และที่นั่งคงเหลือ
เรียงตามอัตราการสอบผ่านจากน้อยไปมาก
WITH course_results AS (
    SELECT sec.section_id,
           c.course_name,
           sec.seat_limit,
           COUNT(e.student_id) AS enrollment_count,
           SUM(CASE WHEN e.grade = 'A' THEN 1 ELSE 0 END) AS grade_a_count,
           SUM(CASE WHEN e.grade IS NOT NULL THEN 1 ELSE 0 END) AS graded_count,
           SUM(CASE
                   WHEN e.grade IS NOT NULL AND e.grade <> 'F' THEN 1
                   ELSE 0
               END) AS passed_count
    FROM SECTION AS sec
    INNER JOIN COURSE AS c
        ON sec.course_id = c.course_id
    LEFT JOIN ENROLLMENT AS e
        ON sec.section_id = e.section_id
    WHERE sec.semester = '2568/1'
    GROUP BY sec.section_id, c.course_name, sec.seat_limit
)
SELECT course_name,
       enrollment_count,
       grade_a_count,
       ROUND(
           100.0 * passed_count / NULLIF(graded_count, 0),
           1
       ) AS pass_rate_percent,
       seat_limit - enrollment_count AS remaining_seats
FROM course_results
ORDER BY pass_rate_percent ASC;
